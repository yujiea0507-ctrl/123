// 我在程式碼中加入這一小段檢查，它會把失蹤的資料自動抓出來！ mounted() { // 檢查有沒有舊資料 const oldData1 = localStorage.getItem('myDailyData'); const oldData2 = localStorage.getItem('dailyData'); // 如果有舊資料，但現在的 dailyData 是空的，就把它抓回來 if (oldData1 && !oldData2) { localStorage.setItem('dailyData', oldData1); this.dailyData = JSON.parse(oldData1); alert("找到舊資料了！已自動恢復！"); } else if (oldData2) { this.dailyData = JSON.parse(oldData2); } }
